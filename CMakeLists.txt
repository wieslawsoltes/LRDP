cmake_minimum_required(VERSION 3.20)
project(LRDP VERSION 0.1.0 LANGUAGES CXX)
include(CTest)
option(LRDP_SANITIZE "Enable address and undefined-behavior sanitizers" OFF)
option(LRDP_BUILD_SERVER "Build the native Linux server" ON)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
add_library(lrdp_protocol src/wire.cpp src/channels.cpp src/clipboard.cpp src/display.cpp src/mcs.cpp src/desktop.cpp src/session.cpp)
target_include_directories(lrdp_protocol PUBLIC include)
target_compile_options(lrdp_protocol PRIVATE -Wall -Wextra -Wpedantic -Wconversion -Wshadow)
if(LRDP_SANITIZE)
  add_compile_options(-fsanitize=address,undefined -fno-omit-frame-pointer)
  add_link_options(-fsanitize=address,undefined)
  target_compile_options(lrdp_protocol PRIVATE -fsanitize=address,undefined -fno-omit-frame-pointer)
endif()
if(BUILD_TESTING)
  add_executable(lrdp_tests tests/protocol_tests.cpp)
  target_link_libraries(lrdp_tests PRIVATE lrdp_protocol)
  target_compile_options(lrdp_tests PRIVATE -Wall -Wextra -Wpedantic)
  add_test(NAME protocol COMMAND lrdp_tests)
endif()

if(LRDP_BUILD_SERVER)
  find_package(OpenSSL REQUIRED)
  add_executable(lrdpd src/main.cpp src/transport.cpp)
  target_link_libraries(lrdpd PRIVATE lrdp_protocol OpenSSL::SSL OpenSSL::Crypto)
  target_compile_options(lrdpd PRIVATE -Wall -Wextra -Wpedantic -Wconversion -Wshadow)
  install(TARGETS lrdpd RUNTIME DESTINATION bin)
  if(BUILD_TESTING)
    find_package(Python3 COMPONENTS Interpreter REQUIRED)
    add_test(NAME tls_session COMMAND ${Python3_EXECUTABLE} ${CMAKE_CURRENT_SOURCE_DIR}/tests/integration.py $<TARGET_FILE:lrdpd>)
    set_tests_properties(tls_session PROPERTIES TIMEOUT 45)
  endif()
endif()
