cmake_minimum_required(VERSION 3.20)
project(LRDP VERSION 0.2.0 LANGUAGES CXX)
include(CTest)
option(LRDP_SANITIZE "Enable address and undefined-behavior sanitizers" OFF)
option(LRDP_BUILD_SERVER "Build the native Linux server" ON)
option(LRDP_X11 "Build the native X11 desktop backend" ON)
option(LRDP_FFMPEG "Build native FFmpeg AVC420 software and hardware encoders" ON)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
if(LRDP_SANITIZE)
  add_compile_options(-fsanitize=address,undefined -fno-omit-frame-pointer)
  add_link_options(-fsanitize=address,undefined)
endif()
find_package(Threads REQUIRED)
add_library(lrdp_protocol src/wire.cpp src/channels.cpp src/clipboard.cpp src/display.cpp
  src/mcs.cpp src/capabilities.cpp src/monitor_layout.cpp src/desktop.cpp src/session.cpp src/session_updates.cpp src/session_input.cpp
  src/session_channels.cpp src/session_graphics.cpp src/session_audio.cpp src/graphics.cpp
  src/pointer.cpp src/video_worker.cpp src/audio_format.cpp src/audio_output.cpp src/audio_input.cpp)
target_include_directories(lrdp_protocol PUBLIC include)
target_link_libraries(lrdp_protocol PUBLIC Threads::Threads)
target_compile_options(lrdp_protocol PRIVATE -Wall -Wextra -Wpedantic -Wconversion -Wshadow)
add_library(lrdp_capture src/capture_frame.cpp)
target_link_libraries(lrdp_capture PUBLIC lrdp_protocol)
target_compile_options(lrdp_capture PRIVATE -Wall -Wextra -Wpedantic -Wconversion -Wshadow)
if(BUILD_TESTING)
  foreach(name IN ITEMS protocol graphics audio transport_queue monitor_layout pointer)
    add_executable(lrdp_${name}_tests tests/${name}_tests.cpp)
    target_link_libraries(lrdp_${name}_tests PRIVATE lrdp_protocol)
    add_test(NAME ${name} COMMAND lrdp_${name}_tests)
  endforeach()
  add_executable(lrdp_capture_tests tests/capture_tests.cpp)
  target_link_libraries(lrdp_capture_tests PRIVATE lrdp_capture)
  add_test(NAME capture_layout COMMAND lrdp_capture_tests)
endif()
if(LRDP_BUILD_SERVER)
  find_package(OpenSSL REQUIRED)
  find_package(PkgConfig QUIET)
  add_executable(lrdpd src/main.cpp src/transport.cpp)
  target_link_libraries(lrdpd PRIVATE lrdp_protocol OpenSSL::SSL OpenSSL::Crypto)
  target_compile_options(lrdpd PRIVATE -Wall -Wextra -Wpedantic -Wconversion -Wshadow)
  install(TARGETS lrdpd RUNTIME DESTINATION bin)
  if(BUILD_TESTING)
    find_package(Python3 COMPONENTS Interpreter REQUIRED)
    foreach(kind IN ITEMS tls gfx)
      if(kind STREQUAL "tls")
        set(fixture integration.py)
      else()
        set(fixture gfx_integration.py)
      endif()
      add_test(NAME ${kind}_session COMMAND ${Python3_EXECUTABLE} ${CMAKE_CURRENT_SOURCE_DIR}/tests/${fixture} $<TARGET_FILE:lrdpd>)
      set_tests_properties(${kind}_session PROPERTIES TIMEOUT 45)
    endforeach()
  endif()
  if(LRDP_X11)
    find_package(X11)
    if(X11_FOUND)
      add_library(lrdp_x11 src/x11_desktop.cpp src/x11_pointer.cpp)
      target_include_directories(lrdp_x11 PRIVATE ${X11_INCLUDE_DIR})
      target_link_libraries(lrdp_x11 PUBLIC lrdp_protocol ${X11_LIBRARIES} ${CMAKE_DL_LIBS})
      target_compile_options(lrdp_x11 PRIVATE -Wall -Wextra -Wpedantic -Wconversion -Wshadow)
      target_link_libraries(lrdpd PRIVATE lrdp_x11)
      target_compile_definitions(lrdpd PRIVATE LRDP_HAVE_X11=1)
      if(BUILD_TESTING)
        find_program(XVFB_EXECUTABLE Xvfb)
        if(XVFB_EXECUTABLE)
          add_executable(lrdp_x11_tests tests/x11_tests.cpp)
          target_link_libraries(lrdp_x11_tests PRIVATE lrdp_x11)
          add_test(NAME x11_desktop COMMAND ${Python3_EXECUTABLE} ${CMAKE_CURRENT_SOURCE_DIR}/tests/xvfb.py $<TARGET_FILE:lrdp_x11_tests>)
          set_tests_properties(x11_desktop PROPERTIES TIMEOUT 40)
          find_program(FREERDP_CLIENT NAMES xfreerdp3 xfreerdp)
          if(FREERDP_CLIENT)
            foreach(mode IN ITEMS bitmap raw)
              set(extra)
              if(mode STREQUAL "raw")
                set(extra --gfx-raw)
              endif()
              add_test(NAME client_${mode} COMMAND ${Python3_EXECUTABLE} ${CMAKE_CURRENT_SOURCE_DIR}/tests/xvfb.py
                ${Python3_EXECUTABLE} ${CMAKE_CURRENT_SOURCE_DIR}/tests/client_interop.py $<TARGET_FILE:lrdpd> ${FREERDP_CLIENT} ${extra})
              set_tests_properties(client_${mode} PROPERTIES TIMEOUT 45 SKIP_RETURN_CODE 77)
            endforeach()
          endif()
        endif()
      endif()
    endif()
  endif()
  if(LRDP_FFMPEG AND PkgConfig_FOUND)
    pkg_check_modules(AVCODEC QUIET IMPORTED_TARGET libavcodec libavutil libswscale)
    if(AVCODEC_FOUND)
      add_library(lrdp_video src/ffmpeg_encoder.cpp)
      target_link_libraries(lrdp_video PUBLIC lrdp_protocol PkgConfig::AVCODEC)
      target_compile_options(lrdp_video PRIVATE -Wall -Wextra -Wpedantic -Wconversion -Wshadow)
      target_link_libraries(lrdpd PRIVATE lrdp_video)
      target_compile_definitions(lrdpd PRIVATE LRDP_HAVE_FFMPEG=1)
      if(BUILD_TESTING)
        add_executable(lrdp_video_tests tests/video_tests.cpp)
        target_link_libraries(lrdp_video_tests PRIVATE lrdp_video)
        add_test(NAME video_codec COMMAND lrdp_video_tests)
        if(FREERDP_CLIENT AND XVFB_EXECUTABLE)
          add_test(NAME client_avc420 COMMAND ${Python3_EXECUTABLE} ${CMAKE_CURRENT_SOURCE_DIR}/tests/xvfb.py
            ${Python3_EXECUTABLE} ${CMAKE_CURRENT_SOURCE_DIR}/tests/client_interop.py $<TARGET_FILE:lrdpd> ${FREERDP_CLIENT} --gfx)
          set_tests_properties(client_avc420 PROPERTIES TIMEOUT 45 SKIP_RETURN_CODE 77)
        endif()
      endif()
    endif()
  endif()
  include(cmake/LrdpHeadless.cmake)
  include(cmake/LrdpExtensions.cmake)
endif()
